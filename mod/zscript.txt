version "4.14"

// Expanding sound ring thrown by the bat: knocks every monster it touches across the room.
class TungWave : Actor
{
	Array<Actor> hit;

	Default
	{
		Radius 22;
		Height 44;
		Speed 46;
		Damage 0;
		Scale 0.35;
		Projectile;
		+RIPPER;
		+NOGRAVITY;
		+BRIGHT;
		RenderStyle "Add";
		Alpha 0.8;
		DamageType "TungBat";
	}

	override int DoSpecialDamage(Actor target, int damage, Name damagetype)
	{
		if (hit.Find(target) != hit.Size() || !target.bShootable) return 0;
		hit.Push(target);
		target.A_StartSound("tung/hit", CHAN_VOICE, CHANF_OVERLAP, 1.0, ATTN_NONE);
		target.A_StartSound("tung/hit", CHAN_ITEM, CHANF_OVERLAP, 0.8, ATTN_NONE);
		target.vel.xy = AngleToVector(angle, 24);
		target.vel.z = 13;
		target.A_QuakeEx(2, 2, 2, 8, 0, 400, "", QF_SCALEDOWN);
		let h = TungHandler(EventHandler.Find("TungHandler"));
		if (h) h.Launched(target);
		TungChip.Burst(target, 6);
		return 35 + random(0, 10);
	}

	States
	{
	Spawn:
		TTSW A 1 A_SetScale(Scale.X + 0.07);
		Loop;
	Death:
		TTSW A 1 A_FadeOut(0.25);
		Wait;
	}
}

// A flying splinter of wood that bounces and fades.
class TungChip : Actor
{
	Default
	{
		Radius 3;
		Height 3;
		Scale 0.55;
		Gravity 0.5;
		BounceType "Doom";
		BounceFactor 0.6;
		BounceCount 3;
		RenderStyle "Translucent";
		Alpha 1.0;
		+NOBLOCKMAP;
		+NOTELEPORT;
		+THRUACTORS;
		+DROPOFF;
	}

	static void Burst(Actor src, int n)
	{
		for (int i = 0; i < n; i++)
		{
			let c = Actor.Spawn("TungChip", src.pos + (0, 0, src.height * 0.5 + frandom(-10, 10)));
			if (!c) continue;
			c.vel = (frandom(-7, 7), frandom(-7, 7), frandom(3, 11));
			c.scale *= frandom(0.7, 1.4);
		}
	}

	States
	{
	Spawn:
		TTSC A 4;
		TTSC B 4;
		TTSC A 4;
		TTSC B 4;
		TTSC A 3 A_FadeOut(0.1);
		TTSC B 3 A_FadeOut(0.1);
		Wait;
	Death:
		TTSC A 4 A_FadeOut(0.12);
		Loop;
	}
}

class TungBat : DoomWeapon
{
	Default
	{
		Weapon.SelectionOrder 100;
		Weapon.SlotNumber 1;
		Weapon.Kickback 0;
		Inventory.PickupMessage "You grabbed the giant wooden bat of Tung Tung Tung Sahur!";
		Inventory.PickupSound "tung/hit";
		Obituary "%o heard the tung tung tung.";
		+WEAPON.MELEEWEAPON;
		+WEAPON.NOALERT;
	}

	action void A_TungSwing()
	{
		A_StartSound("tung/swing", CHAN_WEAPON, CHANF_OVERLAP);
		A_FireProjectile("TungWave", 0, false, 0, 0, 0, 0);
		A_QuakeEx(1, 1, 1, 6, 0, 200, "", QF_SCALEDOWN);
	}

	States
	{
	Spawn:
		TTSP A -1;
		Stop;
	Ready:
		TTSB A 1 A_WeaponReady;
		Loop;
	Deselect:
		TTSB A 1 A_Lower;
		Loop;
	Select:
		TTSB A 1 A_Raise;
		Loop;
	Fire:
		TTSB A 3 A_WeaponOffset(0, 40, WOF_ADD);
		TTSB B 3 A_TungSwing;
		TTSB B 4 A_WeaponOffset(-6, 4, WOF_ADD);
		TTSB A 4 A_WeaponOffset(0, 32);
		TTSB A 6;
		TTSB A 1 A_ReFire;
		Goto Ready;
	}
}

class TungHandler : EventHandler
{
	Array<Actor> flying;
	Array<double> speeds;
	Array<int> since;
	int kills;
	bool given;

	void Launched(Actor a)
	{
		int i = flying.Find(a);
		if (i == flying.Size()) { flying.Push(a); speeds.Push(24); since.Push(level.maptime); }
		else { speeds[i] = 24; since[i] = level.maptime; }
	}

	void Drop(int i)
	{
		flying.Delete(i);
		speeds.Delete(i);
		since.Delete(i);
	}

	override void WorldTick()
	{
		let mo = players[consoleplayer].mo;
		if (!given && mo && level.maptime > 5)
		{
			given = true;
			mo.GiveInventory("TungBat", 1);
			mo.A_SelectWeapon("TungBat");
			Console.MidPrint(null, "TUNG TUNG TUNG SAHUR!");
		}
		// A launched monster that slams into a wall breaks up.
		for (int i = flying.Size() - 1; i >= 0; i--)
		{
			let a = flying[i];
			if (!a || a.health <= 0) { Drop(i); continue; }
			double sp = a.vel.xy.Length();
			if (level.maptime - since[i] > 3 && speeds[i] > 12 && sp < 5 && a.pos.z > a.floorz + 1)
			{
				a.A_StartSound("tung/hit", CHAN_BODY, CHANF_OVERLAP, 1.0, ATTN_NONE);
				TungChip.Burst(a, 8);
				a.DamageMobj(null, mo, 40, 'TungBat');
				Drop(i);
				continue;
			}
			if (a.pos.z <= a.floorz + 1 && level.maptime - since[i] > 3) { Drop(i); continue; }
			speeds[i] = sp;
		}
	}

	override void WorldThingDied(WorldEvent e)
	{
		let t = e.Thing;
		if (!t || !t.bIsMonster || t.DamageTypeReceived != 'TungBat') return;
		t.A_StartSound("tung/chips", CHAN_AUTO, CHANF_OVERLAP, 1.0, ATTN_NONE);
		TungChip.Burst(t, 22);
		t.A_SetRenderStyle(0., STYLE_None);
		kills++;
		if (kills % 3 == 0) Console.MidPrint(null, "TUNG TUNG TUNG SAHUR!");
	}
}
